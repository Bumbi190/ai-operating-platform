 static void RequireLocalDirectory(string path){
  string full=Path.GetFullPath(path);
  Assert(full.Length>=3&&char.IsLetter(full[0])&&full[1]==':'&&full[2]=='\\'&&!full.StartsWith(@"\\")&&full.IndexOf(':',2)<0,"local drive path required");
  Assert(new DriveInfo(Path.GetPathRoot(full)).DriveType==DriveType.Fixed,"fixed local drive required");
  DirectoryInfo current=new DirectoryInfo(full);
  while(current!=null){Assert(current.Exists&&(current.Attributes&FileAttributes.ReparsePoint)==0,"directory absent or reparse point");current=current.Parent;}
 }
 static void RequireLocalFile(string path){
  RequireLocalDirectory(Path.GetDirectoryName(Path.GetFullPath(path)));
  FileAttributes a=File.GetAttributes(path);
  Assert((a&(FileAttributes.Directory|FileAttributes.ReparsePoint))==0,"regular local file required");
 }
 static bool Absent(string path){
  RequireLocalDirectory(Path.GetDirectoryName(path));
  string name=Path.GetFileName(path);
  Assert(name.Length>0&&name.IndexOfAny(new char[]{'*','?'})<0,"exact filename required");
  return Directory.GetFileSystemEntries(Path.GetDirectoryName(path),name,SearchOption.TopDirectoryOnly).Length==0;
 }
