 static void RunObservedTrial(IntPtr expectedSid,bool detached,bool retain,string kind,string sourcePath,string runtime,Result r){
  CrashTrial t=new CrashTrial();t.Kind=kind;t.Nonce=Guid.NewGuid().ToString("N");t.RuntimeImage=runtime;t.DetachedRequested=detached;t.RetainHandleControl=retain;
  t.ArmedMarker=Path.Combine(r.ProfileStorage,t.Nonce+"-armed");t.OwnerFinallyMarker=Path.Combine(r.ProfileStorage,t.Nonce+"-owner-finally");r.Trials.Add(t);
  Require(!File.Exists(t.ArmedMarker)&&!File.Exists(t.OwnerFinallyMarker),"fresh owner marker paths");
  string sidText=expectedSid==IntPtr.Zero?"":new SecurityIdentifier(expectedSid).Value;
  string command="$ErrorActionPreference='Stop';$env:PSModulePath='C:\\Windows\\System32\\WindowsPowerShell\\v1.0\\Modules';$PSModuleAutoLoadingPreference='None';Microsoft.PowerShell.Core\\Import-Module 'C:\\Windows\\System32\\WindowsPowerShell\\v1.0\\Modules\\Microsoft.PowerShell.Utility\\Microsoft.PowerShell.Utility.psd1' -Force;Microsoft.PowerShell.Utility\\Add-Type -TypeDefinition ([IO.File]::ReadAllText("+Ps(sourcePath)+"));[CreativeCrashProbe]::RunOwner("+Ps(sidText)+","+Ps(r.ProfileStorage)+","+Ps(runtime)+","+Ps(t.Nonce)+","+ (detached?"$true":"$false")+")";
  string arguments="\""+OwnerImage+"\" -NoProfile -NonInteractive -EncodedCommand "+Convert.ToBase64String(Encoding.Unicode.GetBytes(command));
  PROCESS owner=new PROCESS();IntPtr observerJob=IntPtr.Zero,buffer=IntPtr.Zero,env=IntPtr.Zero;
  List<PROCESS> members=new List<PROCESS>();bool ownerStarted=false;
  Stopwatch launch=Stopwatch.StartNew();
  try{
   string compilerTemp=Path.Combine(Path.GetDirectoryName(r.ProfileStorage),"LocalState","owner-compiler-"+t.Nonce);Require(!Directory.Exists(compilerTemp),"fresh compiler temp required");Directory.CreateDirectory(compilerTemp);OrdinaryDirectory(compilerTemp);
   env=Marshal.StringToHGlobalUni("LOCALAPPDATA="+Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData)+"\0PATH=C:\\Windows\\System32\0SystemRoot=C:\\Windows\0TEMP="+compilerTemp+"\0TMP="+compilerTemp+"\0WINDIR=C:\\Windows\0\0");
   EX startup=new EX();startup.Startup.Size=104;
   Check(CreateProcessW(OwnerImage,new StringBuilder(arguments),IntPtr.Zero,IntPtr.Zero,false,0x08000400,env,@"C:\Windows\System32",ref startup,out owner),"start own fixed supervisor");ownerStarted=true;t.OwnerPid=owner.Pid;
   Require(string.Equals(ImageName(owner.Process),OwnerImage,StringComparison.OrdinalIgnoreCase),"supervisor image mismatch");t.OwnerToken=InspectToken(owner.Process,IntPtr.Zero);
   CloseHandle(owner.Thread);owner.Thread=IntPtr.Zero;
   while(launch.ElapsedMilliseconds<20000&&WaitForSingleObject(owner.Process,0)==WAIT_TIMEOUT&&!File.Exists(t.ArmedMarker)&&!File.Exists(t.OwnerFinallyMarker))Thread.Sleep(50);
   t.OwnerReadyElapsedMs=launch.ElapsedMilliseconds;
   string errorPath=Path.Combine(r.ProfileStorage,t.Nonce+"-owner-error");if(File.Exists(errorPath))t.OwnerDiagnostic=BoundedMarker(errorPath);
   Require(File.Exists(t.ArmedMarker),"owner did not arm; diagnostic="+t.OwnerDiagnostic);
   string[] armed=BoundedMarker(t.ArmedMarker).Split('|');uint op=0,parentPid=0,childPid=0;long jobValue=0;
   Require(armed.Length==5&&armed[0]==t.Nonce&&uint.TryParse(armed[1],out op)&&op==owner.Pid&&uint.TryParse(armed[2],out parentPid)&&uint.TryParse(armed[3],out childPid)&&parentPid!=0&&childPid!=0&&parentPid!=childPid&&long.TryParse(armed[4],out jobValue)&&jobValue>0,"armed report invalid");
   Require(!File.Exists(t.OwnerFinallyMarker),"owner already left rendezvous");Require(WaitForSingleObject(owner.Process,0)==WAIT_TIMEOUT,"owner must remain live");
   // Duplicate only from the exact CreateProcess-returned supervisor handle, never by a reported supervisor PID.
   Check(DuplicateHandle(owner.Process,new IntPtr(jobValue),new IntPtr(-1),out observerJob,0,false,2),"duplicate job for independent inspection");
   uint flags;Check(GetHandleInformation(observerJob,out flags),"observer job flags");t.ObserverJobHandleNoninheritable=(flags&1)==0;Require(t.ObserverJobHandleNoninheritable,"observer job must not inherit");
   // Owner checked its own source handle before publishing the armed report.
   t.OwnerJobHandleNoninheritable=true;
   buffer=Marshal.AllocHGlobal(144);Check(QueryInformationJobObject(observerJob,9,buffer,144,IntPtr.Zero),"independent policy readback");
   LIMITS read=(LIMITS)Marshal.PtrToStructure(buffer,typeof(LIMITS));Require(read.Basic.Flags==0x2008&&read.Basic.Count==4,"observer policy mismatch");t.ExactPolicyReadBack=true;
   Check(QueryInformationJobObject(observerJob,3,buffer,144,IntPtr.Zero),"independent native members");
   int assigned=Marshal.ReadInt32(buffer,0),count=Marshal.ReadInt32(buffer,4);t.NativeJobProcessCount=count;
   Require(assigned==count&&count>=2&&count<=4,"expected Node pair and at most two console hosts");
   bool parentFound=false,childFound=false;List<uint> seen=new List<uint>();
   for(int i=0;i<count;i++){
    uint pid=(uint)Marshal.ReadInt64(buffer,8+i*8);Require(pid!=0&&!seen.Contains(pid),"invalid duplicate native member");seen.Add(pid);
    PROCESS p=new PROCESS();p.Pid=pid;p.Process=OpenProcess(0x00101001,false,pid);Check(p.Process!=IntPtr.Zero,"hold native member");
    bool belongs;if(!IsProcessInJob(p.Process,observerJob,out belongs)||!belongs){CloseHandle(p.Process);p.Process=IntPtr.Zero;throw new InvalidOperationException("member changed; closed without termination");}
    members.Add(p);ProcessObservation o=new ProcessObservation();o.Pid=pid;o.MembershipVerified=true;o.Image=ImageName(p.Process);t.Members.Add(o);
    if(pid==parentPid){o.Role="Node-parent";Require(!parentFound,"duplicate parent");parentFound=true;}
    else if(pid==childPid){o.Role="Node-child";Require(!childFound,"duplicate child");childFound=true;}
    else{o.Role="console";Require(string.Equals(o.Image,@"C:\Windows\System32\conhost.exe",StringComparison.OrdinalIgnoreCase),"unknown auxiliary image");}
    if(o.Role!="console")Require(string.Equals(o.Image,runtime,StringComparison.OrdinalIgnoreCase),"Node image mismatch");
    o.Token=InspectToken(p.Process,expectedSid,o.Role!="console");
    o.LiveBeforeOwnerTermination=WaitForSingleObject(p.Process,0)==WAIT_TIMEOUT;Require(o.LiveBeforeOwnerTermination,"member must remain live before abrupt termination");
   }
   Require(parentFound&&childFound,"reported pair not native members");
   string parentMarker=BoundedMarker(Path.Combine(r.ProfileStorage,t.Nonce+"-parent.ready")),childMarker=BoundedMarker(Path.Combine(r.ProfileStorage,t.Nonce+"-child.ready"));
   Require(parentMarker==t.Nonce+"|"+parentPid+"|"+childPid+"|"+ExpectedNodeVersion,"independent parent JS marker mismatch");
   Require(childMarker==t.Nonce+"|"+childPid+"|"+ExpectedNodeVersion,"independent child JS marker mismatch");
   t.AllExpectedMembersVerified=true;t.OwnerLiveBeforeTermination=WaitForSingleObject(owner.Process,0)==WAIT_TIMEOUT;Require(t.OwnerLiveBeforeTermination,"owner exited before stimulus");
   Require(launch.ElapsedMilliseconds<25000,"rendezvous deadline exceeded; cannot attribute shutdown to stimulus");
   if(!retain){Check(CloseHandle(observerJob),"close observer job before abrupt owner termination");observerJob=IntPtr.Zero;t.ObserverJobHandleClosedBeforeOwnerTermination=true;}
   Stopwatch stimulus=Stopwatch.StartNew();
   Check(TerminateProcess(owner.Process,AbruptExitCode),"abruptly terminate only the owned supervisor");t.AbruptTerminationRequested=true;
   t.OwnerTerminated=WaitForSingleObject(owner.Process,5000)==0;Require(t.OwnerTerminated,"owner termination wait failed");
   uint exit;Check(GetExitCodeProcess(owner.Process,out exit),"owner actual exit");t.OwnerExitCode=exit;Require(exit==AbruptExitCode,"owner exit must match forced termination");
   t.OwnerFinallyMarkerAbsent=!File.Exists(t.OwnerFinallyMarker);Require(t.OwnerFinallyMarkerAbsent,"owner cleanup marker unexpectedly exists");
   if(retain){
    t.RetainedControlObservationMs=500;Thread.Sleep(t.RetainedControlObservationMs);bool allLive=true;
    for(int i=0;i<members.Count;i++){t.Members[i].LiveWithRetainedHandle=WaitForSingleObject(members[i].Process,0)==WAIT_TIMEOUT;allLive=allLive&&t.Members[i].LiveWithRetainedHandle;}
    t.RetainedHandleKeptMembersAlive=allLive;Require(allLive,"retained-handle control failed to keep members live");
    Check(CloseHandle(observerJob),"close retained control job handle");observerJob=IntPtr.Zero;t.ObserverJobHandleClosedAfterControl=true;
   }
   bool allSignaled=true;for(int i=0;i<members.Count;i++){t.Members[i].ExitedAfterLastHandleClose=WaitForSingleObject(members[i].Process,5000)==0;allSignaled=allSignaled&&t.Members[i].ExitedAfterLastHandleClose;}
   t.PostTerminationElapsedMs=stimulus.ElapsedMilliseconds;t.AllHeldMembersSignaled=allSignaled;
   Require(allSignaled,"not all held members signaled after final job handle release");t.Passed=true;
  }finally{
   // This is fallback cleanup only; Passed was decided before any direct member termination.
   if(observerJob!=IntPtr.Zero){TerminateJobObject(observerJob,2);CloseHandle(observerJob);observerJob=IntPtr.Zero;}
   bool ownerStopped=Stop(ref owner);bool allStopped=true;
   for(int i=0;i<members.Count;i++){PROCESS p=members[i];allStopped=Stop(ref p)&allStopped;}
   bool corroborated=!ownerStarted||t.AllExpectedMembersVerified;
   if(!corroborated&&File.Exists(t.OwnerFinallyMarker)){try{corroborated=BoundedMarker(t.OwnerFinallyMarker)==t.Nonce+"|1";}catch{}}
   r.AllProcessesStopped=(ownerStopped&allStopped&corroborated)&r.AllProcessesStopped;
   if(buffer!=IntPtr.Zero)Marshal.FreeHGlobal(buffer);if(env!=IntPtr.Zero)Marshal.FreeHGlobal(env);
  }

 public static Result Run(string sourcePath){
  string stdioMode="ignore",runtimeLocation="profile";
  Result r=new Result();r.SelectedStdioMode=stdioMode;r.RuntimeLocation=runtimeLocation;r.ProfileName="omnira.p36.crash."+Guid.NewGuid().ToString("N");IntPtr sid=IntPtr.Zero,folder=IntPtr.Zero;
  try{
   Require(IntPtr.Size==8&&Marshal.SizeOf(typeof(LIMITS))==144&&Marshal.SizeOf(typeof(EX))==112&&Marshal.SizeOf(typeof(CAPS))==24,"unsupported x64 ABI");
   Require(!new WindowsPrincipal(WindowsIdentity.GetCurrent()).IsInRole(WindowsBuiltInRole.Administrator),"ordinary user required");InspectToken(new IntPtr(-1),IntPtr.Zero);
   Require(stdioMode=="ignore"||stdioMode=="inherit","unsupported explicit stdio diagnostic mode");
   Require(runtimeLocation=="installed"||runtimeLocation=="profile","unsupported explicit runtime location");
   OrdinaryDirectory(Path.GetDirectoryName(NodeImage));Require((File.GetAttributes(NodeImage)&(FileAttributes.ReparsePoint|FileAttributes.Directory))==0,"fixed Node image must be regular");
   using(FileStream f=File.OpenRead(NodeImage))using(System.Security.Cryptography.SHA256 hash=System.Security.Cryptography.SHA256.Create()){r.NodeObservedSha256=BitConverter.ToString(hash.ComputeHash(f)).Replace("-","").ToLowerInvariant();}
   Hr(CreateAppContainerProfile(r.ProfileName,"Omnira supervisor crash probe","Temporary abrupt owned-supervisor termination diagnostic",IntPtr.Zero,0,out sid),"create own profile");r.ProfileCreated=true;r.ProfileSid=new SecurityIdentifier(sid).Value;
   Hr(GetAppContainerFolderPath(r.ProfileSid,out folder),"own profile storage");r.ProfileStorage=Marshal.PtrToStringUni(folder);Marshal.FreeCoTaskMem(folder);folder=IntPtr.Zero;
   string expected=Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),"Packages",r.ProfileName,"AC");
   Require(string.Equals(Path.GetFullPath(r.ProfileStorage),Path.GetFullPath(expected),StringComparison.OrdinalIgnoreCase),"unexpected profile storage");OrdinaryDirectory(r.ProfileStorage);
   r.RuntimeImagePath=NodeImage;r.RuntimeObservedSha256=r.NodeObservedSha256;
   if(runtimeLocation=="profile"){
    string directory=Path.Combine(r.ProfileStorage,"runtime");Directory.CreateDirectory(directory);OrdinaryDirectory(directory);
    r.RuntimeImagePath=Path.Combine(directory,"node-readiness.exe");Require(!File.Exists(r.RuntimeImagePath),"runtime staging target must be fresh");
    using(FileStream source=File.OpenRead(NodeImage))using(FileStream target=new FileStream(r.RuntimeImagePath,FileMode.CreateNew,FileAccess.Write,FileShare.None)){source.CopyTo(target);target.Flush(true);}r.RuntimeStaged=true;
    Require((File.GetAttributes(r.RuntimeImagePath)&(FileAttributes.ReparsePoint|FileAttributes.Directory))==0,"staged runtime must be regular");
    using(FileStream f=File.OpenRead(r.RuntimeImagePath))using(System.Security.Cryptography.SHA256 hash=System.Security.Cryptography.SHA256.Create()){r.RuntimeObservedSha256=BitConverter.ToString(hash.ComputeHash(f)).Replace("-","").ToLowerInvariant();}
    r.StagedRuntimeHashMatches=r.RuntimeObservedSha256==r.NodeObservedSha256;Require(r.StagedRuntimeHashMatches,"staged runtime hash mismatch");
    System.Security.AccessControl.FileSecurity acl=File.GetAccessControl(r.RuntimeImagePath);
    foreach(System.Security.AccessControl.FileSystemAccessRule rule in acl.GetAccessRules(true,true,typeof(SecurityIdentifier))){
     if(rule.IdentityReference.Value==r.ProfileSid&&rule.AccessControlType==System.Security.AccessControl.AccessControlType.Allow&&(rule.FileSystemRights&System.Security.AccessControl.FileSystemRights.ReadAndExecute)==System.Security.AccessControl.FileSystemRights.ReadAndExecute)r.RuntimeExpectedPackageSidReadExecuteAllowObserved=true;
    }
    Require(r.RuntimeExpectedPackageSidReadExecuteAllowObserved,"fresh AC fixture did not inherit expected package read/execute allow");

   }
   RegularSource(sourcePath);r.SourceSha256=HashFile(sourcePath);
   RunObservedTrial(IntPtr.Zero,false,true,"retained-handle-control",sourcePath,NodeImage,r);
   RunObservedTrial(IntPtr.Zero,false,false,"ordinary-normal-before",sourcePath,NodeImage,r);
   RunObservedTrial(IntPtr.Zero,true,false,"ordinary-detached-before",sourcePath,NodeImage,r);
   RunObservedTrial(sid,false,false,"AppContainer-normal",sourcePath,r.RuntimeImagePath,r);
   RunObservedTrial(sid,true,false,"AppContainer-detached",sourcePath,r.RuntimeImagePath,r);
   RunObservedTrial(IntPtr.Zero,false,false,"ordinary-normal-after",sourcePath,NodeImage,r);
   RunObservedTrial(IntPtr.Zero,true,false,"ordinary-detached-after",sourcePath,NodeImage,r);
  }catch(Exception e){r.Error=e.Message;}
  finally{
   if(folder!=IntPtr.Zero)Marshal.FreeCoTaskMem(folder);
   if(r.ProfileCreated&&r.AllProcessesStopped){
    for(int i=0;i<3;i++){if(DeleteAppContainerProfile(r.ProfileName)==0){r.ProfileDeleteApiSucceeded=true;break;}Thread.Sleep(200);}
    if(r.ProfileDeleteApiSucceeded){try{
     r.ProfileStorageAbsenceCorroborated=Directory.GetFileSystemEntries(Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),"Packages"),r.ProfileName,SearchOption.TopDirectoryOnly).Length==0;
     using(Microsoft.Win32.RegistryKey k=Microsoft.Win32.Registry.CurrentUser.OpenSubKey(@"Software\Classes\Local Settings\Software\Microsoft\Windows\CurrentVersion\AppContainer\Mappings\"+r.ProfileSid)){r.ProfileMappingAbsent=k==null;}
    }catch{}}
   }
   if(sid!=IntPtr.Zero)FreeSid(sid);r.ObservedAtUtc=DateTime.UtcNow.ToString("O");
  }
  if(!r.AllProcessesStopped||(r.ProfileCreated&&(!r.ProfileDeleteApiSucceeded||!r.ProfileStorageAbsenceCorroborated||!r.ProfileMappingAbsent)))r.Error=(r.Error==null?"":r.Error+"; ")+"cleanup unverified";
  return r;
 }
